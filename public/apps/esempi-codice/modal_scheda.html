<!-- 5. SCRIPT JAVASCRIPT PER APERTURA / CHIUSURA -->
<script>
function openCustomHtmlModal() {
  var modal = document.getElementById('custom-html-modal');
  if (modal) {
    modal.classList.add('active');
    document.body.style.overflow = 'hidden';
  }
}

function closeCustomHtmlModal() {
  var modal = document.getElementById('custom-html-modal');
  if (modal) {
    modal.classList.remove('active');
    document.body.style.overflow = '';
  }
}

window.openCustomHtmlModal = openCustomHtmlModal;
window.closeCustomHtmlModal = closeCustomHtmlModal;
window.triggerModalOpen = openCustomHtmlModal;

document.addEventListener('keydown', function(e) {
  if (e.key === 'Escape') closeCustomHtmlModal();
});
</script>
