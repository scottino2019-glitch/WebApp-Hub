    (function() {
      function sendLog(type, args) {
        try {
          const serialized = Array.from(args).map(arg => {
            if (typeof arg === 'object') {
              try { return JSON.stringify(arg); } catch(e) { return String(arg); }
            }
            return String(arg);
          }).join(' ');

          window.parent.postMessage({
            source: 'menu-studio-iframe',
            type: type,
            message: serialized,
            timestamp: new Date().toLocaleTimeString()
          }, '*');
        } catch (err) {}
      }

      const originalLog = console.log;
      const originalWarn = console.warn;
      const originalError = console.error;
      const originalInfo = console.info;

      console.log = function(...args) {
        sendLog('log', args);
        originalLog.apply(console, args);
      };
      console.warn = function(...args) {
        sendLog('warn', args);
        originalWarn.apply(console, args);
      };
      console.error = function(...args) {
        sendLog('error', args);
        originalError.apply(console, args);
      };
      console.info = function(...args) {
        sendLog('info', args);
        originalInfo.apply(console, args);
      };

      window.onerror = function(message, source, lineno, colno, error) {
        sendLog('error', [`Errore JS (linea ${lineno}): ${message}`]);
        return false;
      };
    })();

const canvas = document.getElementById('canvasArea');

canvas.addEventListener('contextmenu', (e) => {
  e.preventDefault();

  const mouseX = e.clientX;
  const mouseY = e.clientY;
  const menuWidth = 240;
  const menuHeight = 220;

  // Evita collisioni con i bordi dello schermo
  const x = (mouseX + menuWidth > window.innerWidth) ? mouseX - menuWidth : mouseX;
  const y = (mouseY + menuHeight > window.innerHeight) ? mouseY - menuHeight : mouseY;

  contextMenu.style.left = `${x}px`;
  contextMenu.style.top = `${y}px`;
  contextMenu.style.display = 'block';
});

// Chiudi cliccando altrove
document.addEventListener('click', (e) => {
  if (!contextMenu.contains(e.target)) {
    contextMenu.style.display = 'none';
  }
});

// Gestione selezione azioni
